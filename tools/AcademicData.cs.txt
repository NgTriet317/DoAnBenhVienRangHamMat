    static DateTime ScenarioMorning(string employee,string room,DateTime cutoff)
    {
        var date=Scalar<DateTime?>("""
            SELECT MAX(v.NgayLam) FROM dbo.LICH_LAM_VIEC v
            WHERE v.MaNV=@employee AND v.MaPhong=@room AND v.MaCa=N'DL26_CA_SANG' AND v.TrangThai=N'Đã xác nhận'
              AND DATEADD(hour,12,CONVERT(datetime2,v.NgayLam))<=@cutoff
            """,new{employee,room,cutoff});
        Require(date.HasValue,"No elapsed working morning for synthetic receipt: "+employee);
        return date!.Value;
    }
