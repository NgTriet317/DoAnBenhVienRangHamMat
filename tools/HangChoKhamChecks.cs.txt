    static void Tx(Action<IDbTransaction> run){using var db=Open();using var tx=db.BeginTransaction(IsolationLevel.Serializable);try{run(tx);}finally{if(tx.Connection is not null)tx.Rollback();}}

    static object? Invoke(object o,string name)=>o.GetType().GetMethod(name,BindingFlags.NonPublic|BindingFlags.Instance)!.Invoke(o,null);

    static void Check(string name,Action run){try{run();Console.WriteLine("PASS: "+name);pass++;}catch(Exception ex){Console.WriteLine("FAIL: "+name+" | "+ex.GetType().Name+": "+ex.Message);fail++;}}
