/// <summary>Điểm chạy công cụ DEMO P5, chuyển chế độ chuẩn bị, xuất bằng chứng hoặc tạo dữ liệu sang P5DemoData.</summary>
internal static class Program
{
    /// <summary>Định tuyến theo đối số và đổi lỗi thao tác thành dòng FAIL cùng mã thoát.</summary>
    /// <param name="args">Máy chủ, database, thư mục kết quả; có thể thêm --prepare hoặc --export ở đầu.</param>
    /// <returns>0 khi chế độ đã chọn hoàn tất; 1 khi đầu vào hoặc thao tác bị lỗi.</returns>
    static int Main(string[] args)
    {
        try
        {
            if(args.Length==4&&args[0]=="--prepare"){P5DemoData.Prepare(args[1],args[2],args[3]);return 0;}
            if(args.Length==4&&args[0]=="--export"){P5DemoData.Export(args[1],args[2],args[3]);return 0;}
            if(args.Length!=3)throw new ArgumentException("Cần server, DB mới và thư mục kết quả.");
            P5DemoData.Create(args[0],args[1],args[2]);return 0;
        }
        catch(Exception ex){Console.WriteLine("FAIL: "+ex.Message);return 1;}
    }
}
