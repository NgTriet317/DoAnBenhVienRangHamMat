    static void ConcurrentReception()
    {
        using var db=new Microsoft.Data.SqlClient.SqlConnection(cs);db.Open();NguCanhNghiepVu actor;
        using(var tx=db.BeginTransaction(IsolationLevel.Serializable))
        {
            Assert(db.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.BENH_NHAN WHERE MaBN=N'SVC_BN'",transaction:tx)==0,"Use a fresh _Concurrent database; persisted fixture already exists");
            ReceptionFixture(tx);
            db.Execute("INSERT dbo.BENH_NHAN(MaBN,HoTenBN,NgaySinh,TrangThai) VALUES(N'SVC_BN2',N'FIXTURE quầy khác','19900101',N'Đang theo dõi')",transaction:tx);
            actor=Login(tx,"NV001","TK001");tx.Commit();
        }
        var principal=Thread.CurrentPrincipal;using var start=new ManualResetEventSlim(false);
        var requests=new[]{ServiceRequest(),ServiceRequest(),ServiceRequest() with{MaBN="SVC_BN2"}};
        var tasks=requests.Select(r=>Task.Run(()=>{Thread.CurrentPrincipal=principal;start.Wait();return new BUS_TIEPNHAN().TiepNhan(r,actor);})).ToArray();
        start.Set();var results=Task.WhenAll(tasks).GetAwaiter().GetResult();
        Assert(results[0].MaPhieu==results[1].MaPhieu&&results[0].MaPhieu!=results[2].MaPhieu&&results[0].SoThuTu!=results[2].SoThuTu,"Concurrent admission duplicated a receipt/queue number");
        Assert(db.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.PHIEU_KHAM_BENH p JOIN dbo.HO_SO_BENH_AN h ON h.MaHS=p.MaHS WHERE h.MaBN IN(N'SVC_BN',N'SVC_BN2') AND p.LoaiKham=N'Dịch vụ' AND p.PhiPhuThuDichVu=30000")==2,"Concurrent receipts do not match both patients");
    }
