using System.Reflection;
using System.Security.Claims;
using System.Windows;
using BUS;
using DAL;
using ET;
using RangHamMat;

// Runs the real CanOpen against an already loaded SQL permission snapshot.
// Database-dependent clinical/report checks remain in the hospital integration suite.
static class Program
{
    static readonly BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic;
    static int passed, failed;

    [STAThread] static int Main()
    {
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary
        {
            Source = new Uri("/RangHamMat.Wpf;component/Themes/Theme.xaml", UriKind.Relative)
        });
        var shell = new MainWindow();
        Thread.CurrentPrincipal = new ClaimsPrincipal(new ClaimsIdentity(new[]
        {
            new Claim(ClaimTypes.NameIdentifier, "TK_MENU_FIXTURE"),
            new Claim(ET.DinhDanhNghiepVu.MaNhanVienClaim, "NV_MENU_FIXTURE")
        }, ET.DinhDanhNghiepVu.KieuXacThuc));
        try
        {
            Check("SQL grant opens personnel for a non-admin role", () =>
            {
                Snapshot(shell, false, "NhanSu");
                Require(CanOpen(shell, "NhanSu"));
            });
            Check("SQL removal hides a default User module", () =>
            {
                Snapshot(shell, false);
                Require(!CanOpen(shell, "TongQuan"));
            });
            Check("Admin follows configured module permissions", () =>
            {
                Snapshot(shell, true);
                Require(!CanOpen(shell, "NhanSu"));
            });
            Check("AI intake follows its SQL grant", () =>
            {
                Snapshot(shell, false, "TroLyAI");
                Require(CanOpen(shell, "TroLyAI"));
                Snapshot(shell, false);
                Require(!CanOpen(shell, "TroLyAI"));
            });
            Check("Unknown modules cannot become navigable through SQL", () =>
            {
                Snapshot(shell, true, "UnknownFixtureModule");
                Require(!CanOpen(shell, "UnknownFixtureModule"));
            });
            Check("Legacy warehouse links resolve to the same permission", () =>
            {
                Snapshot(shell, false, "Kho", "NhapXuatVatTu");
                Require(CanOpen(shell, "VatTu") && CanOpen(shell, "NhapKho") && CanOpen(shell, "XuatKho"));
            });
            Check("Role editor retains the AI intake permission", () =>
                Require(BUS_PHANQUYENVAITRO.CacPhanHe.Count(x => x.Ma == "TroLyAI") == 1));
            Check("Anonymous AI approvals fail before opening SQL", () =>
            {
                Thread.CurrentPrincipal = new ClaimsPrincipal(new ClaimsIdentity());
                DenyAiApproval("NV_MENU_FIXTURE");
            });
            Check("AI approval cannot impersonate another employee", () =>
            {
                Thread.CurrentPrincipal = new ClaimsPrincipal(new ClaimsIdentity(new[]
                {
                    new Claim(ClaimTypes.NameIdentifier, "TK_MENU_FIXTURE"),
                    new Claim(DinhDanhNghiepVu.MaNhanVienClaim, "NV_MENU_FIXTURE"),
                    new Claim("RHM.MaPhien", Guid.NewGuid().ToString("D"))
                }, DinhDanhNghiepVu.KieuXacThuc));
                DenyAiApproval("NV_OTHER_EMPLOYEE");
            });
        }
        finally
        {
            shell.Close();
            app.Shutdown();
        }
        Console.WriteLine($"RESULT: {passed} passed, {failed} failed");
        return failed == 0 ? 0 : 1;
    }

    static void Snapshot(MainWindow shell, bool admin, params string[] modules)
    {
        PhienDangNhap.IsAdmin = admin;
        Set(shell, "quyenPhanHeDaTai", modules.ToHashSet(StringComparer.Ordinal));
        Set(shell, "maTaiKhoanDaTai", "TK_MENU_FIXTURE");
        Set(shell, "phienBanQuyenDaTai", BUS_PHANQUYENVAITRO.PhienBan);
        Set(shell, "lanKiemTraTrangThai", DateTime.UtcNow);
    }
    static void Set(MainWindow shell, string field, object value)
        => typeof(MainWindow).GetField(field, flags)!.SetValue(shell, value);
    static bool CanOpen(MainWindow shell, string module)
        => (bool)typeof(MainWindow).GetMethod("CanOpen", flags)!.Invoke(shell, new object[] { module })!;
    static void Require(bool condition)
    {
        if (!condition) throw new Exception("Access decision did not match the saved SQL permission snapshot.");
    }
    static void DenyAiApproval(string employee)
    {
        var original = Environment.GetEnvironmentVariable("RHM_CONNECTION_STRING");
        try
        {
            // A mistakenly opened connection can only reach a closed loopback port.
            Environment.SetEnvironmentVariable("RHM_CONNECTION_STRING", "Server=127.0.0.1,1;Database=RHM_NoDatabase;Integrated Security=True;Encrypt=True;TrustServerCertificate=True;Connect Timeout=1");
            try { new DAL_AI_TAOPHIEU().taoPhieu(new ET_AI_TAOPHIEU { MaNV = employee }); }
            catch (TuChoiQuyenNghiepVuException) { return; }
            throw new Exception("AI approval accepted an unverified identity.");
        }
        finally { Environment.SetEnvironmentVariable("RHM_CONNECTION_STRING", original); }
    }
    static void Check(string name, Action action)
    {
        try { action(); passed++; Console.WriteLine("PASS: " + name); }
        catch (Exception ex) { failed++; Console.WriteLine("FAIL: " + name + ": " + ex.GetBaseException().Message); }
    }
}
