    static void Flow(IDbTransaction tx)
    {
        Fixture(tx);var a=Login(tx,"NV001","TK001");var k=new BUS_KHAMDIEUTRI(tx);var v=k.LayBanNhap("AD_LS",a);
        v=k.LuuKham(new(){MaLS="AD_LS",DauVanTay=v.DauVanTay,TrieuChung="Mô phỏng",ChanDoan="Kiểm thử",KeHoachXuLy="Mô phỏng",TinhTrang=[new(){LoaiTinhTrang="Mô phỏng",MoTaTinhTrang="Kiểm thử"}]},a);
        var p=k.LuuKeHoach(new(){MaLS="AD_LS",CacBuoc=[new(){MaDV="DV001",DonGiaDuKien=100000,NoiDungBuoc="Kiểm thử"}]},a);
        Assert(p.KeHoach.MaBS=="NV002","Sai bác sĩ kế hoạch");p=k.XacNhanKeHoach(p.KeHoach.MaKHDT,p.DauVanTay,a);
        var service=new BUS_DICHVUTHUCHIEN(tx);Assert(service.LayKhongGianThucHien("AD_LS",a).DuocTao,"Không được thực hiện");
        var s=service.LuuThucHien(new(){MaLS="AD_LS",MaDV="DV001",MaCTKH=p.CacBuoc[0].MaCTKH,MaNVThucHien="NV002",ThoiGianThucHien=DateTime.Now.AddSeconds(-1),SoLuong=1,DonGia=100000,KetQua="Kiểm thử"},a);
        service.XacNhanThucHien(s.DichVu.MaDVThucHien,s.DauVanTay,a);
        var rx=new BUS_DONTHUOC(tx);var r=rx.LuuDon(new(){MaLS="AD_LS",CacDong=[new(){MaThuoc="TH001",LieuDung="Mô phỏng",DuongDung="Mô phỏng",HuongDan="Không dùng điều trị",SoNgayDung=1,SoLuong=1}]},a);
        Assert(r.Don.MaBS=="NV002"&&r.DuocSua,"Sai quyền hoặc bác sĩ đơn");
        var edit=rx.LayBanNhapDon(r.Don.MaDT,a);r=rx.XacNhanDon(r.Don.MaDT,edit.DauVanTay,a);Assert(!r.DuocSua,"Đơn đã ký còn sửa");rx.LayBanInDon(r.Don.MaDT,a);
        k.HoanTatKham("AD_LS",k.LayBanNhap("AD_LS",a).DauVanTay,a);Assert(!k.LayChiTiet("AD_LS",a).DuocSua,"Lượt chốt còn sửa");
        Assert(tx.Connection!.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.NHAT_KY_THAO_TAC WHERE MaTK=N'TK001' AND MaDoiTuong IN(@rx,N'AD_LS',@plan)",new{rx=r.Don.MaDT,plan=p.KeHoach.MaKHDT},tx)>=5,"Thiếu audit admin");
        var cashier=new DAL_THANHTOANTRONGOI(tx);var fee=cashier.LapBangTinh("AD_LS","KHO001",a);Assert(fee.DuocThanhToan,"Không lập được bảng thu: "+fee.ThongBao);
        var request=new YeuCauThanhToan{MaLS="AD_LS",MaKhoCapThuoc="KHO001",MaYeuCau=Guid.NewGuid(),DauVanTayBangTinh=fee.DauVanTay!,SoTienKhachDua=fee.TongTien};
        var paid=cashier.ThanhToan(request,a);Assert(paid.SoTienThu==fee.TongTien&&paid.CacDon.Count==1,"Sai kết quả thu");
        Assert(cashier.ThanhToan(request,a).DaTonTai,"Thu lặp không nhận ra yêu cầu cũ");
        var handover=new DAL_BANGIAOTHUOC(tx);Assert(handover.LayHangDoi(new(MaLS:"AD_LS"),a).TongSo==1,"Thiếu hàng chờ giao");
        handover.LayChiTiet(r.Don.MaDT,a);var given=handover.XacNhanGiao(r.Don.MaDT,a);
        Assert(given.MaNVGiao=="NV001"&&given.DaGiaoHet,"Không giao được thuốc bằng Admin");
        Assert(handover.XacNhanGiao(r.Don.MaDT,a).DaTonTai,"Giao lặp không nhận ra lần cũ");
    }
    static void Fixture(IDbTransaction tx,bool visit=true)
    {
        tx.Connection!.Execute("""
          INSERT dbo.PHIEU_KHAM_BENH(MaPhieu,MaHS,MaPhong,MaBS,MaNVTiepNhan,MaDV,SoThuTu,ThoiGianTiepNhan,LyDoKham,TrangThai)
          SELECT N'AD_PH',MaHS,MaPhong,N'NV002',N'NV005',MaDV,999,DATEADD(minute,-5,SYSDATETIME()),N'Kiểm thử',N'Chờ khám' FROM dbo.PHIEU_KHAM_BENH WHERE MaPhieu=N'PKB001';
          """,transaction:tx);
        Assert(tx.Connection.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.PHIEU_KHAM_BENH WHERE MaPhieu=N'AD_PH'",transaction:tx)==1,"Fixture không tạo phiếu");
        if(visit)tx.Connection.Execute("""
          INSERT dbo.LICH_SU_KHAM_BENH(MaLS,MaPhieu,MaBS,MaPhong,ThoiGianKham,TrieuChung,ChanDoan,KeHoachXuLy,TrangThai)
          SELECT N'AD_LS',MaPhieu,MaBS,MaPhong,DATEADD(minute,-4,SYSDATETIME()),N'Kiểm thử',N'Kiểm thử',N'Kiểm thử',N'Nháp' FROM dbo.PHIEU_KHAM_BENH WHERE MaPhieu=N'AD_PH';
          """,transaction:tx);
    }
    static NguCanhNghiepVu Login(IDbTransaction tx,string employee,string account)
    {
        Thread.CurrentPrincipal=new ClaimsPrincipal(new ClaimsIdentity([new(ClaimTypes.NameIdentifier,account),new(DinhDanhNghiepVu.MaNhanVienClaim,employee)],DinhDanhNghiepVu.KieuXacThuc));
        return new BUS_QUYENNGHIEPVU(tx).LayNguCanh(employee);
    }
    static void InTransaction(Action<IDbTransaction> run){using var db=new Microsoft.Data.SqlClient.SqlConnection(cs);db.Open();using var tx=db.BeginTransaction(IsolationLevel.Serializable);try{run(tx);}finally{if(tx.Connection is not null)tx.Rollback();}}
    static void Denied(Action action){try{action();}catch(TuChoiQuyenNghiepVuException){return;}throw new Exception("Không chặn truy cập trái quyền");}
    static void Assert(bool ok,string reason){if(!ok)throw new Exception(reason);}
    static object? Invoke(object target,string name,params object[] arguments)=>target.GetType().GetMethod(name,System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic)!.Invoke(target,arguments);
    static void Check(string name,Action test){try{test();passed++;Console.WriteLine("PASS: "+name);}catch(Exception ex){failed++;Console.WriteLine("FAIL: "+name+" | "+ex.GetType().Name+": "+ex.Message);}}
}
