    static DocumentPaginator Paginator(FlowDocument document)
    {
        return DonThuocDocument.PhanTrang(document, A4);
    }
    static string Export(DocumentPaginator paginator, string name)
    {
        paginator.ComputePageCount();
        for (int i = 0; i < paginator.PageCount; i++)
        {
            var page = paginator.GetPage(i);
            Require(page != DocumentPage.Missing && page.Size == A4, "Trang in thiếu hoặc sai kích thước.");
            if(i == 0 || i == paginator.PageCount-1) Save(page.Visual, page.Size, name + $"-trang-{i+1}");
        }
        var path = Path.Combine(evidence, name + ".xps");
        if (File.Exists(path)) File.Delete(path);
        using (var xps = new XpsDocument(path, FileAccess.Write)) XpsDocument.CreateXpsDocumentWriter(xps).Write(paginator);
        using var zip = ZipFile.OpenRead(path);
        var pages = zip.Entries.Where(x => x.FullName.EndsWith(".fpage")).Select(entry =>
        {
            using var stream = entry.Open();
            return VisibleText(XDocument.Load(stream));
        }).ToArray();
        pagesByDocument[name] = pages;
        return string.Concat(pages);
    }
    static string VisibleText(XDocument xml)
    {
        // XPS của VisualBrush chứa cả chứng từ trong resources mỗi trang. Chỉ đếm glyph nằm trong Viewbox/Clip.
        var words = new List<string>();
        foreach (var glyph in xml.Descendants().Where(node => node.Name.LocalName == "Glyphs"))
        {
            var brush = glyph.Ancestors().FirstOrDefault(node => node.Name.LocalName == "VisualBrush");
            var boundary = brush ?? xml.Root!;
            var point = new Point(double.Parse((string)glyph.Attribute("OriginX")!, CultureInfo.InvariantCulture),
                double.Parse((string)glyph.Attribute("OriginY")!, CultureInfo.InvariantCulture));
            bool visible = true;
            foreach (var node in new[] { glyph }.Concat(glyph.Ancestors()).TakeWhile(node => node != boundary))
            {
                if (node.Attribute("Clip") is { } clip && !Geometry.Parse(clip.Value).FillContains(point)) visible = false;
                if (node.Attribute("RenderTransform") is { } transform) point = Matrix.Parse(transform.Value).Transform(point);
            }
            var viewport = brush?.Attribute("Viewbox") is { } box ? Rect.Parse(box.Value)
                : new Rect(0, 0, double.Parse((string)xml.Root!.Attribute("Width")!, CultureInfo.InvariantCulture),
                    double.Parse((string)xml.Root.Attribute("Height")!, CultureInfo.InvariantCulture));
            if (visible && viewport.Contains(point)) words.Add((string?)glyph.Attribute("UnicodeString") ?? "");
        }
        return string.Concat(words);
    }
    static void Snapshot(FrameworkElement view, string name, Size size)
    {
        view.Measure(size); view.Arrange(new Rect(size)); view.UpdateLayout();
        Dispatcher.CurrentDispatcher.Invoke(() => { }, DispatcherPriority.ApplicationIdle);
        Save(view, size, name);
    }
    static UserControl LoadLayout(string path)
    {
        var xml = XDocument.Load(path);
        var events = new[] { "Click", "Checked", "Unchecked" };
        foreach (var attribute in xml.Descendants().Attributes().Where(a =>
            a.Name == XName.Get("Class", "http://schemas.microsoft.com/winfx/2006/xaml") || events.Contains(a.Name.LocalName)).ToArray()) attribute.Remove();
        return (UserControl)System.Windows.Markup.XamlReader.Parse(xml.ToString());
    }
    static void Save(Visual visual, Size size, string name)
    {
        var bitmap = new RenderTargetBitmap((int)Math.Ceiling(size.Width), (int)Math.Ceiling(size.Height), 96, 96, PixelFormats.Pbgra32);
        bitmap.Render(visual);
        var png = new PngBitmapEncoder(); png.Frames.Add(BitmapFrame.Create(bitmap));
        using var output = File.Create(Path.Combine(evidence, name + ".png")); png.Save(output);
    }
    static IEnumerable<T> Descendants<T>(DependencyObject node) where T : DependencyObject
    {
        if(node is T match) yield return match;
        for(int i=0; i<VisualTreeHelper.GetChildrenCount(node); i++) foreach(var child in Descendants<T>(VisualTreeHelper.GetChild(node,i))) yield return child;
    }
