    private static void CheckName(string database)
    {if(!System.Text.RegularExpressions.Regex.IsMatch(database,"^RHM_Hospital_Test_P5Demo_[A-Za-z0-9_]+$"))throw new ArgumentException("Only isolated P5 DEMO databases");}
    private static SqlConnection Open(string server,string database)
    {var db=new SqlConnection(new SqlConnectionStringBuilder{DataSource=server,InitialCatalog=database,IntegratedSecurity=true,TrustServerCertificate=true}.ConnectionString);db.Open();return db;}
    public static void Prepare(string server,string database,string outputPath)
    {
        CheckName(database);using var master=Open(server,"master");
        if(master.ExecuteScalar<int>("SELECT COUNT(*) FROM sys.databases WHERE name=@database",new{database})!=0)throw new InvalidOperationException("Database DEMO đã tồn tại; không ghi đè.");
        var output=P5Artifacts.CreateNewDirectory(outputPath);P5Artifacts.Text(Path.Combine(output,Incomplete),database);
    }
    public static void Export(string server,string database,string outputPath)
    {
        CheckName(database);using var db=Open(server,database);RequireDemo(db);
        var output=P5Artifacts.CreateNewDirectory(outputPath);P5Artifacts.Text(Path.Combine(output,Incomplete),database);
        WriteEvidence(db,database,output);Console.WriteLine("DEMO_EXPORT_COMPLETE="+output);
    }
    private static void RequireDemo(SqlConnection db)
    {
        if(db.ExecuteScalar<int>("SELECT COALESCE(TRY_CONVERT(int,value),0) FROM sys.extended_properties WHERE class=0 AND name=N'RHM.Demo.P5'")!=1
            ||db.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.BENH_NHAN WHERE MaBN NOT LIKE N'P5[_]BN[_]%' OR HoTenBN NOT LIKE N'[[]DEMO]%'")!=0)
            throw new InvalidOperationException("Chỉ xuất từ database DEMO đã đánh dấu, không chứa hồ sơ ngoài bộ giả lập.");
    }
