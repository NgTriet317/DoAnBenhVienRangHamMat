    sealed record TableBaseline(string Table, string[] Columns, string[] Keys, long Rows, string Hash);
    static string Q(string name)=>"["+name.Replace("]","]]")+"]";
    static (long Rows,string Hash) Fingerprint(SqlConnection db, TableBaseline table)
    {
        var columns=string.Join(",",table.Columns.Select(Q));var order=string.Join(",",table.Keys.Select(Q));
        var rows=db.ExecuteScalar<long>($"SELECT COUNT_BIG(*) FROM dbo.{Q(table.Table)}");
        var hash=db.ExecuteScalar<byte[]>($"SELECT HASHBYTES('SHA2_256',COALESCE((SELECT {columns} FROM dbo.{Q(table.Table)} ORDER BY {order} FOR JSON PATH,INCLUDE_NULL_VALUES),N'[]'))");
        return(rows, hash is null ? "EMPTY" : Convert.ToHexString(hash));
    }
    static void SaveBaseline(string path)
    {
        using var db=new SqlConnection(cs);db.Open();
        var tables=db.Query<string>("SELECT name FROM sys.tables WHERE is_ms_shipped=0 ORDER BY name").ToArray();
        var snapshots=new List<TableBaseline>();
        foreach(var table in tables)
        {
            var cols=db.Query<string>("SELECT c.name FROM sys.columns c JOIN sys.tables t ON t.object_id=c.object_id WHERE t.name=@table ORDER BY c.column_id",new{table}).ToArray();
            var keys=db.Query<string>("SELECT c.name FROM sys.tables t JOIN sys.indexes i ON i.object_id=t.object_id AND i.is_primary_key=1 JOIN sys.index_columns ic ON ic.object_id=i.object_id AND ic.index_id=i.index_id JOIN sys.columns c ON c.object_id=ic.object_id AND c.column_id=ic.column_id WHERE t.name=@table ORDER BY ic.key_ordinal",new{table}).ToArray();
            Assert(keys.Length>0,"Baseline table must have a stable primary key");
            var snapshot=new TableBaseline(table,cols,keys,0,"");var fingerprint=Fingerprint(db,snapshot);
            snapshots.Add(snapshot with{Rows=fingerprint.Rows,Hash=fingerprint.Hash});
        }
        File.WriteAllText(path,System.Text.Json.JsonSerializer.Serialize(snapshots,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
        Console.WriteLine($"Saved {snapshots.Count} table fingerprints to {path}");
    }
    static void CompareBaseline(string path)
    {
        var tables=System.Text.Json.JsonSerializer.Deserialize<TableBaseline[]>(File.ReadAllText(path))!;
        foreach(var table in tables)
            Check("Preserved legacy rows and columns: "+table.Table,connection=>{
                var actual=Fingerprint(connection,table);
                Assert(actual.Rows==table.Rows && actual.Hash==table.Hash,"Legacy rows changed during migration");
            });
    }

    static void Check(string name, Action<SqlConnection> run)
    {
        try { using var db=new SqlConnection(cs);db.Open();run(db);passed++;Console.WriteLine("PASS: "+name); }
        catch(Exception ex){failed++;Console.WriteLine($"FAIL: {name}: {ex.Message}");}
    }
    static void Case(string name, Action<SqlConnection,SqlTransaction> run) => Check(name, db => {
        using var tx=db.BeginTransaction(IsolationLevel.Serializable);
        try { run(db,tx); } finally { if(tx.Connection is not null)tx.Rollback(); }
    });
    static void Reject(string name, int[] codes, Action<SqlConnection,SqlTransaction> run) => Case(name,(db,tx)=>{
        try {run(db,tx);} catch(SqlException ex) when(codes.Contains(ex.Number)){return;}
        throw new InvalidOperationException("Expected SQL rejection: "+string.Join(",",codes));
    });
