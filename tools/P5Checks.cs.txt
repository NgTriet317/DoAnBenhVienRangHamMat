    static string Fingerprints(SqlConnection db)
    {
        var tables=db.Query<string>("SELECT name FROM sys.tables WHERE is_ms_shipped=0 ORDER BY name");var values=new List<string>();
        foreach(var table in tables)
        {
            var keys=db.Query<string>("SELECT c.name FROM sys.indexes i JOIN sys.index_columns k ON k.object_id=i.object_id AND k.index_id=i.index_id JOIN sys.columns c ON c.object_id=k.object_id AND c.column_id=k.column_id WHERE i.object_id=OBJECT_ID(@name) AND i.is_primary_key=1 ORDER BY k.key_ordinal",new{name="dbo."+table});
            string Q(string value)=>"["+value.Replace("]","]]")+"]";
            var rows=db.ExecuteScalar<string>($"SELECT * FROM dbo.{Q(table)} ORDER BY {string.Join(",",keys.Select(Q))} FOR JSON PATH,INCLUDE_NULL_VALUES");
            values.Add(table+Convert.ToHexString(SHA256.HashData(System.Text.Encoding.UTF8.GetBytes(rows??""))));
        }
        return string.Join("\n",values);
    }
    static void Tx(Action<SqlConnection,SqlTransaction> action){using var db=Open();using var tx=db.BeginTransaction(IsolationLevel.Serializable);try{action(db,tx);}finally{if(tx.Connection is not null)tx.Rollback();}}

    static void Pump(Task task)
    {
        var frame=new DispatcherFrame();var timer=new DispatcherTimer{Interval=TimeSpan.FromMilliseconds(10)};var until=DateTime.UtcNow.AddSeconds(45);
        timer.Tick+=(_,_)=>{if(task.IsCompleted||DateTime.UtcNow>until)frame.Continue=false;};timer.Start();Dispatcher.PushFrame(frame);timer.Stop();if(!task.IsCompleted)throw new TimeoutException();task.GetAwaiter().GetResult();
    }
