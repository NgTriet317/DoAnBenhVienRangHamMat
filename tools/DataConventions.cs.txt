    /// <summary>Lập ánh xạ mã bộ mẫu và tên đăng nhập sang quy ước hiện hành, kiểm để tránh trùng mã đích.</summary>
    /// <returns>Danh sách bảng, cột, mã cũ và mã mới; chưa tự cập nhật dữ liệu SQL.</returns>
    static List<CodeChange> BuildCodeMap()
    {
        var changes=new List<CodeChange>();
        var keys=db.Query<TextColumn>("""
            SELECT t.name TableName,c.name ColumnName FROM sys.tables t JOIN sys.indexes i ON i.object_id=t.object_id AND i.is_primary_key=1
            JOIN sys.index_columns ic ON ic.object_id=i.object_id AND ic.index_id=i.index_id JOIN sys.columns c ON c.object_id=t.object_id AND c.column_id=ic.column_id
            JOIN sys.types y ON y.user_type_id=c.user_type_id
            WHERE y.name IN(N'nvarchar',N'varchar',N'nchar',N'char') AND(SELECT COUNT(*) FROM sys.index_columns x WHERE x.object_id=i.object_id AND x.index_id=i.index_id)=1 ORDER BY t.name
            """,transaction:tx).ToArray();
        foreach(var key in keys)
        {
            var values=db.Query<string>($"SELECT {Q(key.ColumnName)} FROM dbo.{Q(key.TableName)} ORDER BY {Q(key.ColumnName)}",transaction:tx).ToArray();
            string prefix=Regex.Replace(dbconnect.DinhDangMa(key.TableName,1),"[0-9]+$","");
            // DMTT and receipt codes are explicit business identifiers in the existing catalog.
            if(key.TableName=="DM_TINH_TRANG_BENH")prefix="DMTT";
            long next=values.Where(v=>Regex.IsMatch(v,"^"+prefix+"[0-9]+$")).Select(v=>long.Parse(v[prefix.Length..])).DefaultIfEmpty(0).Max();
            var reserved=values.ToHashSet(StringComparer.OrdinalIgnoreCase);
            foreach(var old in values.Where(v=>v.StartsWith("DL26_",StringComparison.Ordinal)||key.TableName=="LICH_SU_KHAM_BENH"&&Regex.IsMatch(v,"^LSKB0(1[3-9]|2[0-9]|3[0-6])$")))
            {
                string id;
                if(key.TableName=="LICH_SU_KHAM_BENH")id="LS"+old[4..];
                else{do{id=prefix+(++next).ToString("D3");}while(reserved.Contains(id));}
                Require(reserved.Add(id),"Code collision: "+id);changes.Add(new(key.TableName,key.ColumnName,old,id));
            }
        }
        changes.Add(new("TAI_KHOAN","TenDangNhap","doan_baoton","user011"));
        changes.Add(new("TAI_KHOAN","TenDangNhap","doan_phauthuat","user012"));
        changes.Add(new("TAI_KHOAN","TenDangNhap","doan_hinhanh","user013"));
        changes.Add(new("TAI_KHOAN","TenDangNhap","doan_phuchoi","user014"));
        Require(changes.Select(c=>c.OldValue).Distinct().Count()==changes.Count,"Duplicate old code");return changes;
    }

    static string Strip(string value)
    {
        string s=Regex.Replace(value,@"\[ĐỒ ÁN[^\]]*\]|\(mô phỏng\)|\(giả lập\)","",RegexOptions.IgnoreCase);
        s=Regex.Replace(s,@"Dữ liệu (giả lập|mô phỏng); giá/lô/serial/chứng từ không phải giao dịch thực tế\.?","",RegexOptions.IgnoreCase);
        s=Regex.Replace(s,@"^(Tình huống|Chỉ định)\s*(mô phỏng)?\s*:\s*","",RegexOptions.IgnoreCase);
        s=Regex.Replace(s,@"\s*trong tình huống mẫu", "",RegexOptions.IgnoreCase).Replace("hồ sơ mẫu","hồ sơ điều trị");
        s=Regex.Replace(s,@"\b(theo tình huống đồ án|tình huống đồ án|đồ án|mô phỏng|giả lập|thực hành)\b","",RegexOptions.IgnoreCase);
        s=Regex.Replace(s,@"\s+"," ").Trim(' ',':',';','-');
        return s.Length==0?"Đã cập nhật thông tin.":s;
    }

    /// <remarks>Tạm ngắt ràng buộc/trigger trong giao dịch rồi bật và kiểm lại; xử lý JSON theo cấu trúc, bỏ qua cột hash/dấu vân tay.</remarks>
    static void ApplyPresentation()
    {
        if(Scalar<int>("SELECT COUNT(*) FROM sys.extended_properties WHERE class=0 AND name=@PresentationMarker",new{PresentationMarker})>0){Console.WriteLine("Presentation migration already applied; checking only");return;}
        Json("counts-before-presentation.json",Counts());FinishExistingVisits();
        var changes=BuildCodeMap();Json("code-map.json",changes);
        var columns=TextColumns();var tables=columns.Select(c=>c.TableName).Distinct().ToArray();
        var triggers=db.Query<string>("SELECT QUOTENAME(OBJECT_SCHEMA_NAME(object_id))+'.'+QUOTENAME(name) FROM sys.triggers WHERE parent_class=1 AND is_ms_shipped=0 AND is_disabled=0",transaction:tx).ToArray();
        foreach(var table in tables)Execute($"ALTER TABLE dbo.{Q(table)} NOCHECK CONSTRAINT ALL");
        foreach(var trigger in triggers)Execute($"DISABLE TRIGGER {trigger} ON {Scalar<string>("SELECT QUOTENAME(OBJECT_SCHEMA_NAME(parent_id))+'.'+QUOTENAME(OBJECT_NAME(parent_id)) FROM sys.triggers WHERE object_id=OBJECT_ID(@trigger)",new{trigger})}");
        Execute("CREATE TABLE #CodeMap(OldValue nvarchar(256) NOT NULL PRIMARY KEY,NewValue nvarchar(256) NOT NULL)");
        foreach(var change in changes)Execute("INSERT #CodeMap VALUES(@OldValue,@NewValue)",change);
        foreach(var column in columns)
            Execute($"UPDATE t SET {Q(column.ColumnName)}=m.NewValue FROM dbo.{Q(column.TableName)} t JOIN #CodeMap m ON CONVERT(nvarchar(max),t.{Q(column.ColumnName)}) COLLATE DATABASE_DEFAULT=m.OldValue");
        var map=changes.ToDictionary(c=>c.OldValue,c=>c.NewValue,StringComparer.Ordinal);
        map["DL26"]="HE_THONG";map["DU_LIEU_DO_AN"]="HE_THONG";
        var token=new Regex("(?<![A-Za-z0-9_])(?:"+string.Join("|",map.Keys.OrderByDescending(k=>k.Length).Select(Regex.Escape))+")(?![A-Za-z0-9_])");
        string Transform(string table,string column,string value)
        {
            // JSON is parsed so embedded codes and display labels stay valid even in audit payloads.
            if(value.TrimStart().StartsWith('{')||value.TrimStart().StartsWith('['))
            {
                try{var node=JsonNode.Parse(value);void Walk(JsonNode? n){if(n is JsonObject obj){foreach(var p in obj.ToArray()){if(p.Value is JsonValue v&&v.TryGetValue<string>(out var s))obj[p.Key]=CleanText(table,p.Key,token.Replace(s,m=>map[m.Value]));else Walk(p.Value);}}else if(n is JsonArray a){for(int i=0;i<a.Count;i++){if(a[i] is JsonValue v&&v.TryGetValue<string>(out var s))a[i]=CleanText(table,column,token.Replace(s,m=>map[m.Value]));else Walk(a[i]);}}}Walk(node);return node!.ToJsonString(new JsonSerializerOptions{Encoder=System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping});}catch(JsonException){}
            }
            return CleanText(table,column,token.Replace(value,m=>map[m.Value]));
        }
        var cleanup=new List<object>();
        foreach(var column in columns)
        {
            // Hashes/passwords and opaque request identifiers remain untouched.
            if(column.ColumnName.Contains("Hash",StringComparison.OrdinalIgnoreCase)||column.ColumnName.Contains("DauVanTay",StringComparison.OrdinalIgnoreCase))continue;
            foreach(var old in db.Query<string>($"SELECT DISTINCT CONVERT(nvarchar(max),{Q(column.ColumnName)}) FROM dbo.{Q(column.TableName)} WHERE {Q(column.ColumnName)} IS NOT NULL",transaction:tx).ToArray())
            {
                var clean=Transform(column.TableName,column.ColumnName,old);
                if(old==clean)continue;
                int rows=Execute($"UPDATE dbo.{Q(column.TableName)} SET {Q(column.ColumnName)}=@clean WHERE CONVERT(nvarchar(max),{Q(column.ColumnName)})=@old",new{clean,old});
                cleanup.Add(new{column.TableName,column.ColumnName,Rows=rows});
            }
        }
        Execute("""
            INSERT dbo.PHONG(MaPhong,TenPhong,MaKhoa,LoaiPhong,TinhTrang,TrangThai) VALUES
            (N'PH016',N'Quầy tiếp nhận',N'KHOA001',N'Quầy tiếp nhận',N'Sẵn sàng',N'Đang hoạt động'),
            (N'PH017',N'Phòng điều phối kho',N'KHOA008',N'Phòng nghiệp vụ kho',N'Sẵn sàng',N'Đang hoạt động'),
            (N'PH018',N'Quầy thanh toán viện phí',N'KHOA010',N'Quầy thanh toán',N'Sẵn sàng',N'Đang hoạt động'),
            (N'PH019',N'Phòng nhân sự',N'KHOA010',N'Phòng hành chính',N'Sẵn sàng',N'Đang hoạt động');
            UPDATE dbo.LICH_LAM_VIEC SET MaPhong=CASE MaNV
              WHEN N'NV001' THEN N'PH010' WHEN N'NV004' THEN N'PH007' WHEN N'NV005' THEN N'PH016'
              WHEN N'NV006' THEN N'PH008' WHEN N'NV007' THEN N'PH017' WHEN N'NV008' THEN N'PH018'
              WHEN N'NV009' THEN N'PH004' WHEN N'NV010' THEN N'PH019' END
            WHERE MaPhong IS NULL AND MaNV IN(N'NV001',N'NV004',N'NV005',N'NV006',N'NV007',N'NV008',N'NV009',N'NV010');
            """);
        foreach(var table in tables)Execute($"ALTER TABLE dbo.{Q(table)} WITH CHECK CHECK CONSTRAINT ALL");
