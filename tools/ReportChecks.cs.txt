    static void Pump(Task task)
    {
        var timer=new DispatcherTimer{Interval=TimeSpan.FromMilliseconds(10)};var frame=new DispatcherFrame();var limit=DateTime.UtcNow.AddSeconds(45);
        timer.Tick+=(_,_)=>{if(task.IsCompleted||DateTime.UtcNow>limit)frame.Continue=false;};timer.Start();
        Dispatcher.PushFrame(frame);timer.Stop();
        if(!task.IsCompleted)throw new TimeoutException("WPF không hoàn tất trong 45 giây.");
        task.GetAwaiter().GetResult();
    }
