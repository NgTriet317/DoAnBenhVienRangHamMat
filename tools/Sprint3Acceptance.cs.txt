    static bool ConfirmOwnDialog()
    {
        var deadline=DateTime.UtcNow.AddSeconds(8);
        while(DateTime.UtcNow<deadline)
        {
            bool answered=false;
            EnumWindows((hwnd,_)=>
            {
                GetWindowThreadProcessId(hwnd,out var pid);if(pid!=(uint)Environment.ProcessId)return true;
                var title=new StringBuilder(256);GetWindowText(hwnd,title,256);
                if(title.ToString()!="Xác nhận")return true;
                answered=PostMessage(hwnd,0x0111,new IntPtr(6),IntPtr.Zero);return false;
            },IntPtr.Zero);
            if(answered)return true;Thread.Sleep(10);
        }
        return false;
    }
    static IEnumerable<Button> FindButtons(DependencyObject node)
    {
        if(node is Button b)yield return b;
        for(int i=0;i<VisualTreeHelper.GetChildrenCount(node);i++)foreach(var child in FindButtons(VisualTreeHelper.GetChild(node,i)))yield return child;
    }
    /// <summary>Chạy tiến trình con giữ heartbeat hoặc tranh chấp cập nhật theo các tệp tín hiệu chung.</summary>
    /// <param name="args">Máy chủ, database thử, chế độ worker và các tệp trao đổi session/tín hiệu/kết quả.</param>

    static int Worker(string[] args)
    {
        if(!System.Text.RegularExpressions.Regex.IsMatch(args[1],@"^RHM_Sprint3_Test_[A-Za-z0-9_]+$"))return 2;
        cs=$"Data Source={args[0]};Initial Catalog={args[1]};Integrated Security=True;TrustServerCertificate=True";
        Environment.SetEnvironmentVariable("RHM_CONNECTION_STRING",cs);
        Principal("NV001","TK001");var actor=new BUS_QUYENNGHIEPVU().LayNguCanh("NV001");var logs=new BUS_NHATKY();var session=logs.BatDauPhien(actor);
        File.WriteAllText(args[3],session.ToString());
        try
        {
            if(args[2]=="--session-worker")
            {
                while(!File.Exists(args[4])){logs.Heartbeat(session,actor);Thread.Sleep(100);}
            }
            else
            {
                if(!SpinWait.SpinUntil(()=>File.Exists(args[4]),TimeSpan.FromSeconds(15)))return 3;
                var row=JsonSerializer.Deserialize<Dictionary<string,JsonElement>>(File.ReadAllText(args[5]))!.ToDictionary(p=>p.Key,p=>(object?)(p.Value.ValueKind==JsonValueKind.Null?null:p.Value.GetString()));
                var values= new Dictionary<string,object?>{{"TenKhoa","RACE_"+Environment.ProcessId},{"DienThoai","0900000000"},{"MoTa","FIXTURE race"}};
                try{new BUS_DANHMUC().Luu("KHOA",values,row,actor,session);File.WriteAllText(args[6],"COMMIT");}
                catch(InvalidOperationException ex) when(ex.Message.Contains("thay đổi bởi người khác")){File.WriteAllText(args[6],"CONFLICT");}
            }
            logs.KetThucPhien(session);return 0;
        }
        catch(Exception ex){Console.Error.WriteLine(ex);return 1;}
    }
    static Process StartWorker(params string[] args)
    {
        var start=new ProcessStartInfo("dotnet"){UseShellExecute=false,CreateNoWindow=true,WindowStyle=ProcessWindowStyle.Hidden};
        start.ArgumentList.Add(Assembly.GetExecutingAssembly().Location);
        var builder=new SqlConnectionStringBuilder(cs);start.ArgumentList.Add(builder.DataSource);start.ArgumentList.Add(builder.InitialCatalog);
        foreach(var arg in args)start.ArgumentList.Add(arg);
        return Process.Start(start)!;
    }
    /// <summary>Kiểm hai phiên độc lập, kết thúc bất thường khi dừng một tiến trình và một cập nhật thắng/một xung đột.</summary>
    /// <remarks>Ghi fixture/nhật ký và tệp phối hợp trong obj/Sprint3Processes; dừng các worker còn chạy trong finally.</remarks>
    static void ProcessAcceptance()
    {
        string dir=Path.GetFullPath("obj/Sprint3Processes/"+Guid.NewGuid().ToString("N"));Directory.CreateDirectory(dir);
        using var db=new SqlConnection(cs);db.Open();Principal("NV001","TK001");var actor=new BUS_QUYENNGHIEPVU().LayNguCanh("NV001");var logs=new BUS_NHATKY();
        using(var one=StartWorker("--session-worker",dir+"/one",dir+"/stop-one"))
        using(var two=StartWorker("--session-worker",dir+"/two",dir+"/stop-two"))
        {
            try
            {
                Check(SpinWait.SpinUntil(()=>File.Exists(dir+"/one")&&File.Exists(dir+"/two"),TimeSpan.FromSeconds(12)),"LOG-04 two application processes started");
                var first=Guid.Parse(File.ReadAllText(dir+"/one"));var second=Guid.Parse(File.ReadAllText(dir+"/two"));
                Check(first!=second,"LOG-04 processes have independent SQL sessions");
                one.Kill(true);one.WaitForExit();
                db.Execute("UPDATE dbo.PHIEN_DANG_NHAP SET HoatDongCuoiUtc=DATEADD(minute,-6,SYSUTCDATETIME()) WHERE MaPhien=@first",new{first});
                var auditSession=logs.BatDauPhien(actor);
                Check(db.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.PHIEN_DANG_NHAP WHERE MaPhien=@first AND TrangThaiPhien=N'Kết thúc bất thường' AND DangXuatUtc IS NULL",new{first})==1,"LOG-05 killed process closes abnormally without fabricated logout");
                Check(db.ExecuteScalar<int>("SELECT COUNT(*) FROM dbo.PHIEN_DANG_NHAP WHERE MaPhien=@second AND TrangThaiPhien=N'Đang mở'",new{second})==1,"LOG-04 survivor process remains open");
                File.WriteAllText(dir+"/stop-two","");Check(two.WaitForExit(12000)&&two.ExitCode==0,"LOG-04 survivor process exits normally");logs.KetThucPhien(auditSession);
            }
            finally{if(!one.HasExited)one.Kill(true);if(!two.HasExited)two.Kill(true);}
        }
        var session=logs.BatDauPhien(actor);
        var original=new BUS_DANHMUC().Luu("KHOA",new Dictionary<string,object?>{{"TenKhoa","RACE_"+Guid.NewGuid().ToString("N")},{"DienThoai","0900000000"},{"MoTa","fixture"}},null,actor,session);
        File.WriteAllText(dir+"/original",JsonSerializer.Serialize(original));logs.KetThucPhien(session);
        using var left=StartWorker("--race-worker",dir+"/left",dir+"/gate",dir+"/original",dir+"/left-result");
        using var right=StartWorker("--race-worker",dir+"/right",dir+"/gate",dir+"/original",dir+"/right-result");
        try
        {
            Check(SpinWait.SpinUntil(()=>File.Exists(dir+"/left")&&File.Exists(dir+"/right"),TimeSpan.FromSeconds(12)),"CAT-04 two editor processes ready");
            File.WriteAllText(dir+"/gate","");Check(left.WaitForExit(12000)&&right.WaitForExit(12000)&&left.ExitCode==0&&right.ExitCode==0,"CAT-04 concurrent processes finish");
            var outcomes=new[]{File.ReadAllText(dir+"/left-result"),File.ReadAllText(dir+"/right-result")};
            Check(outcomes.Count(x=>x=="COMMIT")==1&&outcomes.Count(x=>x=="CONFLICT")==1,"CAT-04 exactly one commit and one conflict across processes");
        }
        finally{if(!left.HasExited)left.Kill(true);if(!right.HasExited)right.Kill(true);}
    }
