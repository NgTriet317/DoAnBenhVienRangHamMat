    /// <summary>

    static void MappedColumns()
    {
        var missing = new List<string>(); int types=0, properties=0;
        using var db = Open();
        foreach(var t in typeof(ET_LICHKHAM).Assembly.GetTypes())
        {
            var table = t.GetCustomAttribute<TableAttribute>(inherit:false); if(table is null) continue;
            types++;
            var columns=db.Query<string>("SELECT name FROM sys.columns WHERE object_id=OBJECT_ID(@name)",new{name="dbo."+table.Name}).ToHashSet(StringComparer.OrdinalIgnoreCase);
            foreach(var p in t.GetProperties().Where(p=>p.CanRead && p.CanWrite && p.GetIndexParameters().Length==0 && p.GetCustomAttribute<WriteAttribute>()?.Write!=false))
            { properties++; if(!columns.Contains(p.Name))missing.Add(table.Name+"."+p.Name); }
        }
        Assert(missing.Count==0, "Missing mapped columns: "+string.Join(", ", missing));
        Console.WriteLine($"  {types} ET tables; {properties} mapped properties checked");
    }

    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>objects.Where</c>, <c>Scalar&lt;int&gt;</c>, <c>Assert</c>.</remarks>
    static void RequiredObjects()

        var missing=objects.Where(n=>Scalar<int>("SELECT COUNT(*) FROM sys.objects o LEFT JOIN sys.triggers t ON t.object_id=o.object_id WHERE o.object_id=OBJECT_ID(@name) AND ISNULL(t.is_disabled,0)=0",new{name="dbo."+n})!=1).ToArray();
        Assert(missing.Length==0,"Missing/disabled V9 objects: "+string.Join(", ",missing));
        Assert(Scalar<int>("SELECT COUNT(*) FROM sys.computed_columns WHERE object_id=OBJECT_ID('dbo.BANG_LUONG') AND name IN('LuongTheoCa','TongLuong') AND is_persisted=1")==2,"Missing persisted salary formulas");

    /// <summary>
    /// Bước kiểm chứng kiểm tra; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>action</c>, <c>Console.WriteLine</c>, <c>Unwrap</c>.</remarks>
    /// <param name="name">Giá trị tên dùng khi thực hiện thao tác; kiểu string.</param>
    /// <param name="action">Giá trị action dùng khi thực hiện thao tác; kiểu Action.</param>
    static void Check(string name, Action action) { try {action(); passed++;Console.WriteLine("PASS "+name);}catch(Exception ex){failed++;Console.WriteLine("FAIL "+name+": "+Unwrap(ex).Message);} }
    /// <summary>
    /// Bước kiểm chứng gỡ lớp ngoại lệ; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>Unwrap</c>.</remarks>
    /// <param name="ex">Giá trị ex dùng khi thực hiện thao tác; kiểu Exception.</param>
    /// <returns>Kết quả kiểu Exception dùng cho bước xử lý tiếp theo.</returns>
    static Exception Unwrap(Exception ex)=>ex is TargetInvocationException && ex.InnerException is not null ? Unwrap(ex.InnerException):ex;
    /// <summary>
    /// Bước kiểm chứng kiểm tra từ chối; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>Check</c>, <c>action</c>, <c>Unwrap</c>.</remarks>
    /// <param name="name">Giá trị tên dùng khi thực hiện thao tác; kiểu string.</param>
    /// <param name="number">Giá trị số dùng khi thực hiện thao tác; kiểu int.</param>
    /// <param name="action">Giá trị action dùng khi thực hiện thao tác; kiểu Action.</param>
    static void Reject(string name,int number,Action action)=>Check(name,()=>{try{action();}catch(Exception ex){if(Unwrap(ex) is SqlException sql && sql.Number==number)return;throw;}throw new Exception("Expected SQL error "+number);});
    /// <summary>
    /// Bước kiểm chứng khẳng định điều kiện; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <param name="condition">Giá trị tình trạng dùng khi thực hiện thao tác; kiểu bool.</param>
    /// <param name="text">Giá trị nội dung dùng khi thực hiện thao tác; kiểu string.</param>
    static void Assert(bool condition,string text){if(!condition)throw new Exception(text);}
    /// <summary>
    /// Bước kiểm chứng mở; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>db.Open</c>.</remarks>
    /// <returns>Kết quả kiểu SqlConnection dùng cho bước xử lý tiếp theo.</returns>

    /// <summary>
    /// Bước kiểm chứng giá trị đơn; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>Open</c>, <c>db.ExecuteScalar&lt;T&gt;</c>.</remarks>
    /// <param name="sql">Giá trị SQL dùng khi thực hiện thao tác; kiểu string.</param>
    /// <param name="args">Giá trị đối số dùng khi thực hiện thao tác; kiểu object?.</param>
    /// <returns>Kết quả kiểu T dùng cho bước xử lý tiếp theo.</returns>
    /// <typeparam name="T">Kiểu dữ liệu dùng cho thao tác tổng quát này.</typeparam>
    static T Scalar<T>(string sql,object? args=null){using var db=Open();return db.ExecuteScalar<T>(sql,args ?? new{admin})!;}
    /// <summary>
    /// Bước kiểm chứng thi hành SQL; chuẩn bị dữ liệu hoặc kiểm tra kết quả theo các điều kiện trong hàm.
    /// </summary>
    /// <remarks>Các bước được phối hợp qua <c>Open</c>, <c>db.Execute</c>.</remarks>
    /// <param name="sql">Giá trị SQL dùng khi thực hiện thao tác; kiểu string.</param>
    static void Exec(string sql){using var db=Open();db.Execute(sql,new{admin},commandTimeout:90);}
    // Sequential connections reuse SqlClient's ambient transaction connection. No Complete: always rollback.
    static void Fixture(Action action,bool calculate=true)
    {

        action();

    /// <summary>

    /// <summary>

    static void VerifySalary()
    {
        var r=Payroll();

            Assert(Convert.ToDecimal(r[name])==value,$"{name}: expected {value}; got {r[name]}");
        var reconciliation=new DAL_BANGLUONG().layDoiSoatTheoKy(First,Last).AsEnumerable().Single(x=>x.Field<string>("MaBL")==r.Field<string>("MaBL"));
