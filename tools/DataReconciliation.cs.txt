using BUS;
using Microsoft.Data.SqlClient;

/// <summary>Công cụ đối soát chỉ đọc qua BUS, xuất số liệu tổng hợp và chi tiết phát hiện bằng P5Artifacts.</summary>
internal static class Program
{
    /// <summary>Kiểm đối số, chạy các quy tắc đối soát và ghi bộ bằng chứng vào thư mục mới.</summary>
    /// <param name="args">Máy chủ SQL, database, thư mục kết quả mới và giới hạn chi tiết mỗi quy tắc từ 1 đến 500.</param>
    /// <returns>0 khi xuất đối soát hoàn tất dù có phát hiện; 1 khi tham số, truy vấn hoặc ghi bằng chứng bị lỗi.</returns>
    static int Main(string[] args)
    {
        try
        {
            if(args.Length!=4||!System.Text.RegularExpressions.Regex.IsMatch(args[1],"^[A-Za-z0-9_]+$"))throw new ArgumentException("Cần server, database, thư mục mới và giới hạn chi tiết.");
            int limit=int.Parse(args[3]);if(limit is <1 or >500)throw new ArgumentOutOfRangeException(nameof(limit));
            Environment.SetEnvironmentVariable("RHM_CONNECTION_STRING",new SqlConnectionStringBuilder{DataSource=args[0],InitialCatalog=args[1],IntegratedSecurity=true,TrustServerCertificate=true,ConnectTimeout=10}.ConnectionString);
            var result=new BUS_DOISOATDULIEU().KiemTra(limit);
            var output=P5Artifacts.CreateNewDirectory(args[2]);P5Artifacts.Audit(output,result);
            Console.WriteLine($"RESULT: {result.CacBang.Count} tables; {result.TongSaiXacDinh} definite findings; {result.TongCanDoiChieu} require review; no database changes");
            foreach(var rule in result.QuyTac.Where(r=>r.TongSo>0))Console.WriteLine($"FINDING {rule.Ma}: {rule.TongSo}; {rule.MucDo}; detail={rule.ChiTiet.Count}; limited={rule.DaGioiHan}");
            Console.WriteLine("AUDIT_COMPLETE="+output);return 0;
        }
        catch(Exception ex){Console.WriteLine("FAIL: "+ex.Message);return 1;}
    }
}
